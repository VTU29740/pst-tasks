class Solution {
    public String dayOfTheWeek(int day, int month, int year) {
        // Days of the week starting from Sunday
        String[] week = {"Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"};

        // Days in each month (non-leap year by default)
        int[] daysInMonth = {31, 28, 31, 30, 31, 30,
                             31, 31, 30, 31, 30, 31};

        // Calculate total days from 1971-01-01 to the given date
        int days = 0;

        // Add days for full years between 1971 and year-1
        for (int y = 1971; y < year; y++) {
            days += isLeapYear(y) ? 366 : 365;
        }

        // Add days for full months in the current year
        for (int m = 1; m < month; m++) {
            days += daysInMonth[m - 1];
            if (m == 2 && isLeapYear(year)) {
                days += 1; // February in leap year
            }
        }

        // Add days in the current month
        days += day - 1;

        // Jan 1, 1971 was a Friday → index 5
        return week[(days + 5) % 7];
    }

    private boolean isLeapYear(int year) {
        return (year % 400 == 0) || (year % 4 == 0 && year % 100 != 0);
    }
}
